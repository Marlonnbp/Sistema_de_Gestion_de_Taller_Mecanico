package gestores;

import modelo.ItemCatalogo;
import modelo.Repuesto;
import modelo.Servicio;

public class Catalogo {

    private static final int CAPACIDAD = 100;

    private final ItemCatalogo[] items;
    private int cantidad;

    public Catalogo() {
        items = new ItemCatalogo[CAPACIDAD];
        cantidad = 0;
    }

    public int getCantidad() {
        return cantidad;
    }

    public boolean estaLleno() {
        return cantidad == CAPACIDAD;
    }

    public boolean existe(String codigo) {
        return leer(codigo) != null;
    }

    public boolean crear(ItemCatalogo item) {
        if (item == null || estaLleno()) {
            return false;
        }
        
        // Evitar duplicados por código y permitir reactivación
        for (int i = 0; i < cantidad; i++) {
            if (items[i].getCodigo().equals(item.getCodigo())) {
                if (!items[i].isActivo()) {
                    items[i] = item; // Sobrescribe y reactiva el ítem
                    return true;
                }
                return false; // Ya existe y está activo
            }
        }
        
        items[cantidad++] = item;
        return true;
    }

    public ItemCatalogo leer(String codigo) {
        if (codigo == null) {
            return null;
        }
        for (int i = 0; i < cantidad; i++) {
            // Solo retorna el ítem si está activo
            if (items[i].isActivo() && items[i].getCodigo().equals(codigo.trim())) {
                return items[i];
            }
        }
        return null;
    }

    public ItemCatalogo[] buscarPorNombre(String texto) {
        if (texto == null || texto.trim().isEmpty()) {
            return new ItemCatalogo[0];
        }
        String criterio = texto.trim().toLowerCase();
        
        // Primero contamos cuántos ítems activos coinciden
        int total = 0;
        for (int i = 0; i < cantidad; i++) {
            if (items[i].isActivo() && items[i].getNombre().toLowerCase().contains(criterio)) {
                total++;
            }
        }
        
        // Luego creamos el arreglo con el tamaño exacto y lo llenamos
        ItemCatalogo[] resultado = new ItemCatalogo[total];
        int k = 0;
        for (int i = 0; i < cantidad; i++) {
            if (items[i].isActivo() && items[i].getNombre().toLowerCase().contains(criterio)) {
                resultado[k++] = items[i];
            }
        }
        return resultado;
    }

    public ItemCatalogo[] listar() {
        // Contar activos para dimensionar el arreglo
        int totalActivos = 0;
        for (int i = 0; i < cantidad; i++) {
            if (items[i].isActivo()) {
                totalActivos++;
            }
        }
        
        ItemCatalogo[] copia = new ItemCatalogo[totalActivos];
        int k = 0;
        for (int i = 0; i < cantidad; i++) {
            if (items[i].isActivo()) {
                copia[k++] = items[i];
            }
        }
        return copia;
    }

    public boolean modificarPrecio(String codigo, double nuevoPrecio) {
        ItemCatalogo item = leer(codigo);
        if (item == null) {
            return false;
        }
        item.setPrecio(nuevoPrecio);
        return true;
    }

    public boolean modificarStock(String codigo, int nuevoStock) {
        ItemCatalogo item = leer(codigo);
        if (!(item instanceof Repuesto)) {
            return false;
        }
        ((Repuesto) item).setStock(nuevoStock);
        return true;
    }

    public boolean actualizar(String codigo, String nuevoNombre, double nuevoPrecio, int nuevoStock) {
        ItemCatalogo item = leer(codigo);
        if (item == null) {
            return false;
        }
        item.setNombre(nuevoNombre);
        item.setPrecio(nuevoPrecio);
        if (item instanceof Repuesto && nuevoStock >= 0) {
            ((Repuesto) item).setStock(nuevoStock);
        }
        return true;
    }

    // Reemplaza eliminar() por darDeBaja() para mantener el historial
    public boolean darDeBaja(String codigo) {
        if (codigo == null) {
            return false;
        }
        for (int i = 0; i < cantidad; i++) {
            if (items[i].getCodigo().equals(codigo.trim()) && items[i].isActivo()) {
                items[i].setActivo(false); // Baja lógica
                return true;
            }
        }
        return false;
    }

    public boolean cargarDesdeLinea(String linea) {
        if (linea == null || linea.trim().isEmpty()) {
            return false;
        }
        String tipo = linea.split(";")[0];
        if (tipo.equals("REPUESTO")) {
            return crear(Repuesto.desdeLinea(linea));
        }
        if (tipo.equals("SERVICIO")) {
            return crear(Servicio.desdeLinea(linea));
        }
        throw new IllegalArgumentException("Tipo de ítem desconocido: " + tipo);
    }

    public String[] toLineas() {
        // Al guardar en archivos, se exportan TODOS (activos e inactivos)
        // para que las órdenes del historial no pierdan su referencia al cargarse.
        String[] lineas = new String[cantidad];
        for (int i = 0; i < cantidad; i++) {
            lineas[i] = items[i].toLinea();
        }
        return lineas;
    }
    
    public ItemCatalogo leerCualquiera(String codigo) {
        if (codigo == null) return null;
        for (int i = 0; i < cantidad; i++) {
            if (items[i].getCodigo().equals(codigo.trim())) {
                return items[i];
            }
        }
        return null;
    }
}

        cantidad++;
        return nueva;
    }

        for (int i = 0; i < cantidad; i++) {

                ordenes[i].getCliente().getDni().equals(criterio) ||
                ordenes[i].getVehiculo().getPlaca().equalsIgnoreCase(criterio)) {
                coincidencias++;
            }
        }
        
        OrdenTrabajo[] resultado = new OrdenTrabajo[coincidencias];
        int indice = 0;
        for (int i = 0; i < cantidad; i++) {
            if (ordenes[i].getCodigo().equalsIgnoreCase(criterio) || 
                ordenes[i].getCliente().getDni().equals(criterio) ||
                ordenes[i].getVehiculo().getPlaca().equalsIgnoreCase(criterio)) {
                resultado[indice++] = ordenes[i];
            }
        }
        return resultado;
    }

    public boolean actualizar(OrdenTrabajo o) {
        for (int i = 0; i < cantidad; i++) {
            if (ordenes[i].getCodigo().equalsIgnoreCase(o.getCodigo())) {
                ordenes[i] = o;
                return true;
            }
        }
        return false;
    }

    public boolean anular(String codigo) {

            if (ordenes[i].getCodigo().equalsIgnoreCase(codigo)) {
                if (ordenes[i].estaActiva()) {
                    ordenes[i].anular();
                    return true;
                }
                return false;
            }
        }
        return false;
    }

    public boolean tieneOrdenesActivas(Cliente c) {

package modelo;

public abstract class ItemCatalogo {

    private final String codigo;
    private String nombre;
    private double precio;
    private boolean activo; // Agregado para la baja lógica

    public ItemCatalogo(String codigo, String nombre, double precio) {
        if (codigo == null || codigo.trim().isEmpty()) {
            throw new IllegalArgumentException("El código es obligatorio.");
        }
        this.codigo = codigo.trim();
        setNombre(nombre);
        setPrecio(precio);
        this.activo = true; // Por defecto nace activo
    }

    // ---------- Getters ----------

    public String getCodigo() { return codigo; }
    public String getNombre() { return nombre; }
    public double getPrecio() { return precio; }
    public boolean isActivo() { return activo; }

    // ---------- Setters ----------

    public void setNombre(String nombre) {
        if (nombre == null || nombre.trim().isEmpty()) {
            throw new IllegalArgumentException("El nombre es obligatorio.");
        }
        this.nombre = nombre.trim();
    }

    public void setPrecio(double precio) {
        if (precio < 0) {
            throw new IllegalArgumentException("El precio no puede ser negativo.");
        }
        this.precio = precio;
    }

    public void setActivo(boolean activo) {
        this.activo = activo;
    }

    public double calcularImporte(int cantidad) {
        if (cantidad <= 0) {
            throw new IllegalArgumentException("La cantidad debe ser mayor que cero.");
        }
        return precio * cantidad;
    }

    public abstract String getTipo();

    public abstract String toLinea();

    protected String lineaBase() {
        // Se añade el estado 'activo' para que se guarde en el archivo .txt
        return codigo + ";" + nombre + ";" + precio + ";" + activo;
    }

    // ---------- Object ----------

    @Override
    public boolean equals(Object obj) {
        if (this == obj) return true;
        if (!(obj instanceof ItemCatalogo)) return false;
        ItemCatalogo otro = (ItemCatalogo) obj;
        return codigo.equals(otro.codigo);
    }

    @Override
    public int hashCode() {
        return codigo.hashCode();
    }

    @Override
    public String toString() {
        return getTipo() + " [" + codigo + "] " + nombre + " - S/ " + String.format("%.2f", precio) + (activo ? "" : " (INACTIVO)");
    }
}.

        if (cantidadItems < items.length) {
            items[cantidadItems] = new ItemCotizacion(item, cantidad);
            cantidadItems++;
        }
    }

    public double calcularSubtotal() {
        double subtotal = 0;
        for (int i = 0; i < cantidadItems; i++) {
            subtotal += items[i].calcularSubtotal();
        }
        return subtotal;
    }

    public double calcularIGV() {
        return calcularSubtotal() * 0.18; // IGV 18% para Perú
    }

    public double calcularTotal() {
        return calcularSubtotal() + calcularIGV();
    }

    public boolean tieneStockDisponibleParaAprobar() {
        for (int i = 0; i < cantidadItems; i++) {
            if (items[i].getReferencia() instanceof Repuesto) {
                Repuesto r = (Repuesto) items[i].getReferencia();
                if (!r.hayStock(items[i].getCantidad())) {
                    return false;
                }
            }
        }
        return true;
    }

    public void descontarStockRepuestos() {
        for (int i = 0; i < cantidadItems; i++) {
            if (items[i].getReferencia() instanceof Repuesto) {
                Repuesto r = (Repuesto) items[i].getReferencia();
                r.descontarStock(items[i].getCantidad());
            }
        }
    }

    public void reponerStockRepuestos() {
        for (int i = 0; i < cantidadItems; i++) {
            if (items[i].getReferencia() instanceof Repuesto) {
                Repuesto r = (Repuesto) items[i].getReferencia();
                r.reponerStock(items[i].getCantidad());
            }
        }
    }

    public void cambiarEstado(EstadoOrden nuevo) {
        if (nuevo == EstadoOrden.APROBADA) {
            aprobar();
            return;
        }
        if (nuevo == EstadoOrden.ANULADA) {
            anular();
            return;
        }
        this.estado = nuevo;
    }

    public void aprobar() {
        // Al aprobar la orden, se retiran físicamente los repuestos del almacén
        if (this.estado != EstadoOrden.APROBADA && this.estado != EstadoOrden.LISTA_PARA_ENTREGA && this.estado != EstadoOrden.FACTURADA) {
            descontarStockRepuestos();
        }
        this.estado = EstadoOrden.APROBADA;
    }

    public void anular() {
        // Si ya había sido aprobada o estaba lista, se reponen los repuestos de vuelta al almacén
        if (this.estado == EstadoOrden.APROBADA || this.estado == EstadoOrden.LISTA_PARA_ENTREGA) {
            reponerStockRepuestos();
        }
        this.estado = EstadoOrden.ANULADA;
    }

    public boolean estaActiva() {
        return this.estado != EstadoOrden.FACTURADA && this.estado != EstadoOrden.ANULADA;
    }

package modelo;

public class Repuesto extends ItemCatalogo {

    private int stock;

    public Repuesto(String codigo, String nombre, double precio, int stock) {
        super(codigo, nombre, precio);
        setStock(stock);
    }

    public int getStock() { return stock; }

    public void setStock(int stock) {
        if (stock < 0) {
            throw new IllegalArgumentException("El stock no puede ser negativo.");
        }
        this.stock = stock;
    }

    public boolean hayStock(int cantidad) {
        return cantidad > 0 && stock >= cantidad;
    }

    public void descontarStock(int cantidad) {
        if (cantidad <= 0) {
            throw new IllegalArgumentException("La cantidad debe ser mayor que cero.");
        }
        if (cantidad > stock) {
            throw new IllegalArgumentException("Stock insuficiente de " + getNombre() + ".");
        }
        stock -= cantidad;
    }

    public void reponerStock(int cantidad) {
        if (cantidad <= 0) {
            throw new IllegalArgumentException("La cantidad debe ser mayor que cero.");
        }
        stock += cantidad;
    }

    @Override
    public String getTipo() {
        return "REPUESTO";
    }

    @Override
    public String toLinea() {
        return getTipo() + ";" + lineaBase() + ";" + stock;
    }

    public static Repuesto desdeLinea(String linea) {
        String[] p = linea.split(";");
        // Nuevo formato: TIPO(0); CODIGO(1); NOMBRE(2); PRECIO(3); ACTIVO(4); STOCK(5)
        if (p.length != 6 || !p[0].equals("REPUESTO")) {
            throw new IllegalArgumentException("Línea de repuesto inválida: " + linea);
        }
        Repuesto r = new Repuesto(p[1], p[2], Double.parseDouble(p[3]), Integer.parseInt(p[5]));
        r.setActivo(Boolean.parseBoolean(p[4]));
        return r;
    }

    @Override
    public String toString() {
        return super.toString() + " | Stock: " + stock;
    }

    private void guardarItem() {
        String tipo = (String) cmbTipo.getSelectedItem();
        String codigo = txtCodigo.getText().trim();
        String nombre = txtNombre.getText().trim();
        String strPrecio = txtPrecio.getText().trim();

        if (codigo.isEmpty() || nombre.isEmpty() || strPrecio.isEmpty()) {
            JOptionPane.showMessageDialog(this, "Complete todos los campos obligatorios.", "Validación", JOptionPane.WARNING_MESSAGE);
            return;
        }

        double precio;
        try {
            precio = Double.parseDouble(strPrecio);
            if (precio <= 0) throw new NumberFormatException();
        } catch (NumberFormatException ex) {
            JOptionPane.showMessageDialog(this, "El precio debe ser un número positivo.", "Validación", JOptionPane.WARNING_MESSAGE);
            return;
        }

        ItemCatalogo nuevo;
        if ("REPUESTO".equals(tipo)) {
            String strStock = txtStock.getText().trim();
            int stock;
            try {
                stock = Integer.parseInt(strStock);
                if (stock < 0) throw new NumberFormatException();
            } catch (NumberFormatException ex) {
                JOptionPane.showMessageDialog(this, "El stock debe ser un entero mayor o igual a 0.", "Validación", JOptionPane.WARNING_MESSAGE);
                return;
            }
            nuevo = new Repuesto(codigo, nombre, precio, stock);
        } else {
            nuevo = new Servicio(codigo, nombre, precio);
        }

        boolean ok = catalogo.crear(nuevo);
        if (ok) {
            JOptionPane.showMessageDialog(this, "Ítem agregado al catálogo con éxito.", "Éxito", JOptionPane.INFORMATION_MESSAGE);
            limpiarFormulario();
            cargarTablaCatalogo();
        } else {
            JOptionPane.showMessageDialog(this, "El código ya existe o el catálogo está lleno.", "Error", JOptionPane.ERROR_MESSAGE);
        }
    }

    private void actualizarItem() {
        String codigo = txtCodigo.getText().trim();
        String nombre = txtNombre.getText().trim();
        String strPrecio = txtPrecio.getText().trim();
        String tipo = (String) cmbTipo.getSelectedItem();

        if (codigo.isEmpty() || nombre.isEmpty() || strPrecio.isEmpty()) {
            JOptionPane.showMessageDialog(this, 
                    "Seleccione un ítem del catálogo o ingrese sus datos para actualizar.", 
                    "Validación", 
                    JOptionPane.WARNING_MESSAGE);
            return;
        }

        // RF-15 Precondición: El ítem existe
        ItemCatalogo existente = catalogo.leer(codigo);
        if (existente == null) {
            JOptionPane.showMessageDialog(this, 
                    "El ítem con código '" + codigo + "' no existe en el catálogo activo.", 
                    "Ítem no encontrado", 
                    JOptionPane.ERROR_MESSAGE);
            return;
        }

        double nuevoPrecio;
        try {
            nuevoPrecio = Double.parseDouble(strPrecio);
            if (nuevoPrecio < 0) throw new NumberFormatException();
        } catch (NumberFormatException ex) {
            JOptionPane.showMessageDialog(this, "El precio debe ser un número mayor o igual a 0.", "Validación", JOptionPane.WARNING_MESSAGE);
            return;
        }

        int nuevoStock = -1;
        if ("REPUESTO".equals(tipo)) {
            String strStock = txtStock.getText().trim();
            try {
                nuevoStock = Integer.parseInt(strStock);
                if (nuevoStock < 0) throw new NumberFormatException();
            } catch (NumberFormatException ex) {
                JOptionPane.showMessageDialog(this, "El stock de repuesto debe ser un número entero mayor o igual a 0.", "Validación", JOptionPane.WARNING_MESSAGE);
                return;
            }
        }

        // RF-15 Proceso: Actualizar valores
        boolean actualizado = catalogo.actualizar(codigo, nombre, nuevoPrecio, nuevoStock);
        if (actualizado) {
            // RF-15 Postcondición: Muestra valores actualizados (cotizaciones previas congelan sus precios)
            JOptionPane.showMessageDialog(this, 
                    "Ítem '" + codigo + "' actualizado correctamente.\n" +
                    "(Las cotizaciones ya emitidas conservan sus precios congelados intactos).", 
                    "Actualización Exitosa", 
                    JOptionPane.INFORMATION_MESSAGE);
            cargarTablaCatalogo();
            limpiarFormulario();
        } else {
            JOptionPane.showMessageDialog(this, "No se pudo actualizar el ítem.", "Error", JOptionPane.ERROR_MESSAGE);
        }
    }

            } else {
                txtStock.setText("");
                txtStock.setEnabled(false);
            }
        }

    private boolean tieneFacturaEmitida(OrdenTrabajo orden) {

    private void emitirFacturaDialogo() {
        OrdenTrabajo[] todasLasOrdenes = gestorOrdenes.listarTodos();
        
        // REGLA DE NEGOCIO: Solo se puede emitir factura si el estado de la orden es FACTURADA y no tiene factura previa
        int cantFacturables = 0;
        for (OrdenTrabajo o : todasLasOrdenes) {
            if (o.getEstado() == EstadoOrden.FACTURADA && !tieneFacturaEmitida(o)) {
                cantFacturables++;
            }
        }

        if (cantFacturables == 0) {
            JOptionPane.showMessageDialog(this, 
                    "Regla de Negocio:\n" +
                    "Solo se puede emitir una factura si el estado de la orden es FACTURADA y está pendiente de comprobante.\n" +
                    "Cambie el estado de la orden a FACTURADA en la pestaña 'Órdenes de Trabajo' para poder emitir la factura.", 
                    "Regla de Negocio - Emisión de Factura", 
                    JOptionPane.WARNING_MESSAGE);
            return;
        }

        JComboBox<String> comboOrdenes = new JComboBox<>();
        OrdenTrabajo[] facturables = new OrdenTrabajo[cantFacturables];
        int k = 0;
        for (OrdenTrabajo o : todasLasOrdenes) {
            if (o.getEstado() == EstadoOrden.FACTURADA && !tieneFacturaEmitida(o)) {
                facturables[k++] = o;
                comboOrdenes.addItem(o.getCodigo() + " - " + o.getCliente().getNombres() + " (Total: S/. " + String.format("%.2f", o.calcularTotal()) + ") [Estado: FACTURADA]");
            }
        }

        int opt = JOptionPane.showConfirmDialog(this, comboOrdenes, "Seleccione la orden FACTURADA a emitir comprobante:", JOptionPane.OK_CANCEL_OPTION);
        if (opt == JOptionPane.OK_OPTION) {
            int idx = comboOrdenes.getSelectedIndex();
            OrdenTrabajo seleccionada = facturables[idx];

            if (seleccionada.getCantidadItems() == 0) {
                JOptionPane.showMessageDialog(this, "La orden no tiene ningún servicio o repuesto cotizado.", "Aviso", JOptionPane.WARNING_MESSAGE);
                return;
            }

            int numFacturas = historialFacturas.listarTodos().length + 1;
            String numeroFactura = "FAC-" + String.format("%04d", numFacturas);
            Factura f = new Factura(numeroFactura);
            f.emitir(seleccionada);

            boolean registrada = historialFacturas.crear(f);
            if (registrada) {
                JOptionPane.showMessageDialog(this, "Factura " + numeroFactura + " emitida con éxito.", "Facturación", JOptionPane.INFORMATION_MESSAGE);
                cargarTablaHistorial();
            } else {
                JOptionPane.showMessageDialog(this, "No se pudo registrar la factura en el historial.", "Error", JOptionPane.ERROR_MESSAGE);
            }
        }
    }

    public void cargarTablaHistorial() {
        modeloTablaHistorial.setRowCount(0);
        OrdenTrabajo[] todasLasOrdenes = gestorOrdenes.listarTodos();
        Factura[] facturas = historialFacturas.listarTodos();

        for (OrdenTrabajo o : todasLasOrdenes) {
            // Solo se muestran las órdenes que ya concluyeron su ciclo: FACTURADAS o ANULADAS
            if (o.getEstado() == EstadoOrden.FACTURADA || o.getEstado() == EstadoOrden.ANULADA) {
                Factura fAsociada = null;
                for (Factura f : facturas) {
                    if (f.getOrden() != null && f.getOrden().getCodigo().equalsIgnoreCase(o.getCodigo())) {
                        fAsociada = f;
                        break;
                    }
                }

                String numFactura = (fAsociada != null) ? fAsociada.getNumero() : (o.getEstado() == EstadoOrden.FACTURADA ? "Pendiente Comprobante" : "N/A - Cancelada");
                String fecha = (fAsociada != null && fAsociada.getFechaEmision() != null)
                        ? sdf.format(fAsociada.getFechaEmision())
                        : sdf.format(o.getFechaIngreso());
                String nomCliente = (o.getCliente() != null) ? o.getCliente().getNombres() : "-";
                String placaVehiculo = (o.getVehiculo() != null) ? o.getVehiculo().getPlaca() : "-";

                modeloTablaHistorial.addRow(new Object[]{
                        o.getCodigo(),
                        o.getEstado().toString(),
                        numFactura,
                        fecha,
                        nomCliente,
                        placaVehiculo,
                        String.format("%.2f", o.calcularTotal())
                });
            }
        }
    }

    private void verTicketFactura() {
        int fila = tablaHistorial.getSelectedRow();
        if (fila < 0) {
            JOptionPane.showMessageDialog(this, "Seleccione una orden del historial para ver su comprobante.", "Validación", JOptionPane.WARNING_MESSAGE);
            return;
        }

        String estado = modeloTablaHistorial.getValueAt(fila, 1).toString();
        if ("ANULADA".equalsIgnoreCase(estado)) {
            JOptionPane.showMessageDialog(this, 
                    "Esta orden fue anulada/cancelada, por lo que no cuenta con un comprobante de pago emitido.", 
                    "Sin Comprobante", 
                    JOptionPane.WARNING_MESSAGE);
            return;
        }

        String codigoOrden = modeloTablaHistorial.getValueAt(fila, 0).toString();
        Factura[] lista = historialFacturas.listarTodos();
        Factura f = null;
        for (Factura fac : lista) {
            if (fac.getOrden() != null && fac.getOrden().getCodigo().equalsIgnoreCase(codigoOrden)) {
                f = fac;
                break;
            }
        }

        if (f == null) {
            JOptionPane.showMessageDialog(this, 
                    "Esta orden está en estado FACTURADA pero aún no se ha generado su comprobante formal.\n" +
                    "Use el botón 'Emitir Factura para Orden...' para generar el número de factura y su comprobante.", 
                    "Comprobante no generado", 
                    JOptionPane.INFORMATION_MESSAGE);
            return;
        }

        OrdenTrabajo o = f.getOrden();

        StringBuilder sb = new StringBuilder();
        sb.append("========================================\n");
        sb.append("      TALLER MECÁNICO - COMPROBANTE     \n");
        sb.append("========================================\n");
        sb.append("Factura N°:   ").append(f.getNumero()).append("\n");
        sb.append("Fecha:        ").append(sdf.format(f.getFechaEmision())).append("\n");
        sb.append("Cliente:      ").append(o.getCliente().getNombres()).append(" (DNI: ").append(o.getCliente().getDni()).append(")\n");
        sb.append("Vehículo:     ").append(o.getVehiculo().getPlaca()).append(" - ").append(o.getVehiculo().getColor()).append("\n");
        sb.append("Orden N°:     ").append(o.getCodigo()).append("\n");
        sb.append("Motivo:       ").append(o.getMotivoIngreso()).append("\n");
        sb.append("----------------------------------------\n");
        sb.append("DETALLE DE SERVICIOS Y REPUESTOS:\n");
        for (int i = 0; i < o.getCantidadItems(); i++) {
            ItemCotizacion item = o.getItems()[i];
            sb.append(String.format(" - %-20s x%-2d S/. %7.2f\n",
                    item.getReferencia().getNombre(),
                    item.getCantidad(),
                    item.calcularSubtotal()));
        }
        sb.append("----------------------------------------\n");
        sb.append(String.format("SUBTOTAL:                 S/. %10.2f\n", f.getSubtotal()));
        sb.append(String.format("IGV (18%%):                S/. %10.2f\n", f.getIgv()));
        sb.append(String.format("TOTAL A PAGAR:            S/. %10.2f\n", f.getTotal()));
        sb.append("========================================\n");
        sb.append("       ¡Gracias por su preferencia!     \n");

        JTextArea areaTicket = new JTextArea(sb.toString());
        areaTicket.setFont(new Font("Monospaced", Font.PLAIN, 12));
        areaTicket.setEditable(false);

        JOptionPane.showMessageDialog(this, new JScrollPane(areaTicket), "Ticket de Venta - " + f.getNumero(), JOptionPane.PLAIN_MESSAGE);
    }

    private void darDeBajaCliente() {

        if (dni.isEmpty()) {
            JOptionPane.showMessageDialog(this, "Ingrese o seleccione el DNI del cliente a dar de baja.", "Validación", JOptionPane.WARNING_MESSAGE);
            return;

        Cliente cliente = gestorClientes.leer(dni);

            JOptionPane.showMessageDialog(this, "No se encontró ningún cliente activo con el DNI indicado.", "Error", JOptionPane.ERROR_MESSAGE);
            return;
        }

        // El sistema verifica que el cliente no tenga órdenes activas
        if (gestorOrdenes != null && gestorOrdenes.tieneOrdenesActivas(cliente)) {
            JOptionPane.showMessageDialog(this,
                    "No se puede dar de baja al cliente '" + cliente.getNombres() + 
                    "' porque tiene órdenes de trabajo activas en el taller.",
                    "Baja Rechazada",
                    JOptionPane.WARNING_MESSAGE);
            return;
        }

        int confirmacion = JOptionPane.showConfirmDialog(this,
                "¿Está seguro de dar de baja al cliente con DNI " + dni + " (" + cliente.getNombres() + ")?",
                "Confirmar Baja Lógica",
                JOptionPane.YES_NO_OPTION);

        if (confirmacion == JOptionPane.YES_OPTION) {
            boolean dadoDeBaja = gestorClientes.darDeBaja(dni);

                JOptionPane.showMessageDialog(this, "Cliente dado de baja correctamente.", "Éxito", JOptionPane.INFORMATION_MESSAGE);

    private void crearOrden() {
        String dni = txtDniCliente.getText().trim();
        String placa = txtPlacaVehiculo.getText().trim();
        String motivo = txtMotivo.getText().trim();

        if (dni.isEmpty() || placa.isEmpty() || motivo.isEmpty()) {
            JOptionPane.showMessageDialog(this, "Debe ingresar DNI, Placa y Motivo.", "Validación", JOptionPane.WARNING_MESSAGE);
            return;
        }

        Cliente c = gestorClientes.leer(dni);
        if (c == null) {
            JOptionPane.showMessageDialog(this, "Cliente con DNI " + dni + " no existe o está inactivo.", "Error", JOptionPane.ERROR_MESSAGE);
            return;
        }

        Vehiculo v = gestorVehiculos.leer(placa);
        if (v == null) {
            JOptionPane.showMessageDialog(this, "Vehículo con placa " + placa + " no existe o está inactivo.", "Error", JOptionPane.ERROR_MESSAGE);
            return;
        }

        if (gestorOrdenes.tieneOrdenActiva(v)) {
            JOptionPane.showMessageDialog(this, "El vehículo " + placa + " ya tiene una orden activa pendiente en el taller.", "Vehículo en Taller", JOptionPane.WARNING_MESSAGE);
            return;
        }

        OrdenTrabajo nueva = gestorOrdenes.crear(c, v, motivo);
        if (nueva != null) {
            JOptionPane.showMessageDialog(this, "Orden creada con éxito: " + nueva.getCodigo(), "Éxito", JOptionPane.INFORMATION_MESSAGE);
            txtDniCliente.setText("");
            txtPlacaVehiculo.setText("");
            txtMotivo.setText("");
            cargarTablaOrdenes();
        } else {
            JOptionPane.showMessageDialog(this, "No se pudo crear la orden: límite de capacidad alcanzado.", "Error", JOptionPane.ERROR_MESSAGE);
        }
    }

    public void cargarTablaOrdenes() {

    private boolean esEstadoActivo(EstadoOrden estado) {
        return estado == EstadoOrden.EN_ESPERA ||
               estado == EstadoOrden.EN_DIAGNOSTICO ||
               estado == EstadoOrden.COTIZADA ||
               estado == EstadoOrden.APROBADA ||
               estado == EstadoOrden.LISTA_PARA_ENTREGA;
    }

    private void mostrarEnTabla(OrdenTrabajo[] lista) {
        modeloTablaOrdenes.setRowCount(0);
        for (OrdenTrabajo o : lista) {
            // Solo se muestran órdenes activas en taller; las FACTURADAS o ANULADAS van al Historial
            if (esEstadoActivo(o.getEstado())) {
                modeloTablaOrdenes.addRow(new Object[]{
                        o.getCodigo(),
                        sdf.format(o.getFechaIngreso()),
                        o.getCliente().getNombres(),
                        o.getVehiculo().getPlaca(),
                        o.getMotivoIngreso(),
                        o.getEstado(),
                        String.format("%.2f", o.calcularTotal())
                });
            }
        }
        modeloTablaItems.setRowCount(0);
        lblTotalOrden.setText("Total Orden: S/. 0.00   ");
    }

    private OrdenTrabajo obtenerOrdenSeleccionada() {
        int fila = tablaOrdenes.getSelectedRow();
        if (fila < 0) return null;
        String codigo = modeloTablaOrdenes.getValueAt(fila, 0).toString();
        OrdenTrabajo[] busqueda = gestorOrdenes.leer(codigo);
        if (busqueda.length > 0) return busqueda[0];
        return null;
    }

    private void cargarItemsDeOrdenSeleccionada() {
        OrdenTrabajo orden = obtenerOrdenSeleccionada();
        modeloTablaItems.setRowCount(0);
        if (orden == null) {
            lblTotalOrden.setText("Total Orden: S/. 0.00   ");
            return;
        }

        ItemCotizacion[] items = orden.getItems();
        for (int i = 0; i < orden.getCantidadItems(); i++) {
            ItemCotizacion ic = items[i];
            String tipo = (ic.getReferencia() instanceof Servicio) ? "Servicio" : "Repuesto";
            modeloTablaItems.addRow(new Object[]{
                    ic.getReferencia().getCodigo(),
                    ic.getReferencia().getNombre(),
                    tipo,
                    ic.getCantidad(),
                    String.format("%.2f", ic.getPrecioUnitario()),
                    String.format("%.2f", ic.calcularSubtotal())
            });
        }

        lblTotalOrden.setText(String.format("Subtotal: S/. %.2f | IGV (18%%): S/. %.2f | Total: S/. %.2f   ",
                orden.calcularSubtotal(), orden.calcularIGV(), orden.calcularTotal()));
    }

    private void agregarItemAOrden() {
        OrdenTrabajo orden = obtenerOrdenSeleccionada();
        if (orden == null) {
            JOptionPane.showMessageDialog(this, "Seleccione primero una orden de la lista.", "Validación", JOptionPane.WARNING_MESSAGE);
            return;
        }

        if (!orden.estaActiva()) {
            JOptionPane.showMessageDialog(this, "No se pueden agregar ítems a una orden finalizada o anulada.", "Aviso", JOptionPane.WARNING_MESSAGE);
            return;
        }

        // Precondición: La orden debe estar en estado EN_DIAGNOSTICO (o COTIZADA para seguir añadiendo ítems)
        if (orden.getEstado() != EstadoOrden.EN_DIAGNOSTICO && orden.getEstado() != EstadoOrden.COTIZADA) {
            JOptionPane.showMessageDialog(this,
                    "Para cotizar servicios o repuestos, la orden debe estar en estado 'EN_DIAGNOSTICO'.\n" +
                    "Estado actual de la orden " + orden.getCodigo() + ": " + orden.getEstado() + ".\n" +
                    "Cambie primero el estado a 'EN_DIAGNOSTICO' con el botón '⚙️ Cambiar Estado'.",
                    "Estado no válido para cotizar",
                    JOptionPane.WARNING_MESSAGE);
            return;
        }

        ItemCatalogo[] disponibles = catalogo.listar();
        if (disponibles.length == 0) {
            JOptionPane.showMessageDialog(this, "El catálogo está vacío. Agregue servicios o repuestos en la pestaña Catálogo.", "Catálogo Vacío", JOptionPane.WARNING_MESSAGE);
            return;
        }

        // Crear diálogo para elegir ítem y cantidad
        JComboBox<String> comboItems = new JComboBox<>();
        for (ItemCatalogo item : disponibles) {
            comboItems.addItem(item.getCodigo() + " - " + item.getNombre() + " (S/. " + item.getPrecio() + ")");
        }

        JSpinner spinnerCantidad = new JSpinner(new SpinnerNumberModel(1, 1, 100, 1));

        JPanel panelDialogo = new JPanel(new GridLayout(2, 2, 5, 5));
        panelDialogo.add(new JLabel("Seleccionar Ítem:"));
        panelDialogo.add(comboItems);
        panelDialogo.add(new JLabel("Cantidad:"));
        panelDialogo.add(spinnerCantidad);

        int opcion = JOptionPane.showConfirmDialog(this, panelDialogo, "Elaborar Cotización - Orden " + orden.getCodigo(), JOptionPane.OK_CANCEL_OPTION);
        if (opcion == JOptionPane.OK_OPTION) {
            int indice = comboItems.getSelectedIndex();
            ItemCatalogo seleccionado = disponibles[indice];
            int cant = (Integer) spinnerCantidad.getValue();

            // Validar existencia de stock al cotizar si es un repuesto
            if (seleccionado instanceof Repuesto) {
                Repuesto rep = (Repuesto) seleccionado;
                if (!rep.hayStock(cant)) {
                    JOptionPane.showMessageDialog(this,
                            "Stock insuficiente en almacén para cotizar este repuesto.\n" +
                            "Repuesto: " + rep.getNombre() + "\n" +
                            "Stock disponible: " + rep.getStock() + " unidades\n" +
                            "Cantidad requerida: " + cant + " unidades",
                            "Stock Insuficiente",
                            JOptionPane.WARNING_MESSAGE);
                    return;
                }
            }

            orden.agregarItem(seleccionado, cant);

            // RF-17 Postcondición: La orden queda cotizada y pendiente de aprobación
            orden.cambiarEstado(EstadoOrden.COTIZADA);

            JOptionPane.showMessageDialog(this, 
                    "Ítem agregado con éxito a la cotización.\n" +
                    "La orden " + orden.getCodigo() + " ahora se encuentra en estado 'COTIZADA' (pendiente de aprobación).", 
                    "Cotización Elaborada", 
                    JOptionPane.INFORMATION_MESSAGE);
            cargarItemsDeOrdenSeleccionada();
            cargarTablaOrdenes();
        }
    }

    private void cambiarEstadoOrden() {
        OrdenTrabajo orden = obtenerOrdenSeleccionada();
        if (orden == null) {
            JOptionPane.showMessageDialog(this, "Seleccione una orden de la lista.", "Validación", JOptionPane.WARNING_MESSAGE);
            return;
        }

        JComboBox<EstadoOrden> comboEstados = new JComboBox<>(EstadoOrden.values());
        comboEstados.setSelectedItem(orden.getEstado());

        int res = JOptionPane.showConfirmDialog(this, comboEstados, "Cambiar estado de orden " + orden.getCodigo(), JOptionPane.OK_CANCEL_OPTION);
        if (res == JOptionPane.OK_OPTION) {
            EstadoOrden nuevoEstado = (EstadoOrden) comboEstados.getSelectedItem();

            // Momento A: Al APROBAR la orden se descuenta el stock de repuestos
            if (nuevoEstado == EstadoOrden.APROBADA && orden.getEstado() != EstadoOrden.APROBADA) {
                if (!orden.tieneStockDisponibleParaAprobar()) {
                    JOptionPane.showMessageDialog(this,
                            "No se puede aprobar la orden porque uno o más repuestos cotizados ya no cuentan con stock suficiente en el almacén.",
                            "Stock Insuficiente al Aprobar",
                            JOptionPane.ERROR_MESSAGE);
                    return;
                }
                orden.aprobar();
                JOptionPane.showMessageDialog(this,
                        "Orden " + orden.getCodigo() + " APROBADA exitosamente.\n" +
                        "Se han retirado y descontado del stock del almacén los repuestos cotizados.",
                        "Orden Aprobada - Stock Actualizado",
                        JOptionPane.INFORMATION_MESSAGE);
            } else {
                orden.cambiarEstado(nuevoEstado);
            }

            cargarTablaOrdenes();
            cargarItemsDeOrdenSeleccionada();
        }
    }

    private void anularOrden() {
        OrdenTrabajo orden = obtenerOrdenSeleccionada();
        if (orden == null) {
            JOptionPane.showMessageDialog(this, "Seleccione la orden a anular.", "Validación", JOptionPane.WARNING_MESSAGE);
            return;
        }

        int resp = JOptionPane.showConfirmDialog(this, "¿Está seguro de anular la orden " + orden.getCodigo() + "?", "Confirmar Anulación", JOptionPane.YES_NO_OPTION);
        if (resp == JOptionPane.YES_OPTION) {
            boolean estabaAprobada = (orden.getEstado() == EstadoOrden.APROBADA || orden.getEstado() == EstadoOrden.LISTA_PARA_ENTREGA);
            boolean anulada = gestorOrdenes.anular(orden.getCodigo());
            if (anulada) {
                String msj = "Orden " + orden.getCodigo() + " anulada.";
                if (estabaAprobada) {
                    msj += "\nLos repuestos asignados fueron devueltos automáticamente al stock del almacén.";
                }
                JOptionPane.showMessageDialog(this, msj, "Aviso de Anulación", JOptionPane.INFORMATION_MESSAGE);
                cargarTablaOrdenes();
                cargarItemsDeOrdenSeleccionada();
            } else {
                JOptionPane.showMessageDialog(this, "No se pudo anular la orden (quizás ya está cerrada o facturada).", "Error", JOptionPane.ERROR_MESSAGE);
            }
        }
    }
}

    private void registrarVehiculo() {
        String placa = txtPlaca.getText().trim();
        String strAnio = txtAnio.getText().trim();
        String color = txtColor.getText().trim();
        String strKm = txtKilometraje.getText().trim();
        String dniDueño = txtDniPropietario.getText().trim();

        if (placa.isEmpty() || strAnio.isEmpty() || color.isEmpty() || strKm.isEmpty() || dniDueño.isEmpty()) {
            JOptionPane.showMessageDialog(this, "Todos los campos son obligatorios.", "Validación", JOptionPane.WARNING_MESSAGE);
            return;
        }

        int anio;
        int kilometraje;
        try {
            anio = Integer.parseInt(strAnio);
            kilometraje = Integer.parseInt(strKm);
        } catch (NumberFormatException ex) {
            JOptionPane.showMessageDialog(this, "Año y Kilometraje deben ser valores numéricos enteros.", "Validación", JOptionPane.WARNING_MESSAGE);
            return;
        }

        Cliente propietario = gestorClientes.leer(dniDueño);
        if (propietario == null) {
            JOptionPane.showMessageDialog(this, "No existe un cliente activo con el DNI: " + dniDueño + ". Regístrelo primero en la pestaña Clientes.", "Propietario no encontrado", JOptionPane.ERROR_MESSAGE);
            return;
        }

        Vehiculo nuevo = new Vehiculo(placa, anio, color, kilometraje, propietario);
        boolean registrado = gestorVehiculos.crear(nuevo);

        if (registrado) {
            JOptionPane.showMessageDialog(this, "Vehículo registrado correctamente.", "Éxito", JOptionPane.INFORMATION_MESSAGE);
            cargarTabla();
            limpiarFormulario();
        } else {
            JOptionPane.showMessageDialog(this, "No se pudo registrar: La placa ya existe o se alcanzó la capacidad máxima.", "Error", JOptionPane.ERROR_MESSAGE);
        }
    }

    private void actualizarVehiculo() {
        String placa = txtPlaca.getText().trim();
        String strAnio = txtAnio.getText().trim();
        String color = txtColor.getText().trim();
        String strKm = txtKilometraje.getText().trim();
        String dniDueño = txtDniPropietario.getText().trim();

        if (placa.isEmpty()) {
            JOptionPane.showMessageDialog(this, "Seleccione un vehículo e ingrese la placa a actualizar.", "Validación", JOptionPane.WARNING_MESSAGE);
            return;
        }

        int anio;
        int kilometraje;
        try {
            anio = Integer.parseInt(strAnio);
            kilometraje = Integer.parseInt(strKm);
        } catch (NumberFormatException ex) {
            JOptionPane.showMessageDialog(this, "Año y Kilometraje deben ser numéricos.", "Validación", JOptionPane.WARNING_MESSAGE);
            return;
        }

        Cliente propietario = gestorClientes.leer(dniDueño);
        if (propietario == null) {
            JOptionPane.showMessageDialog(this, "No existe cliente con DNI " + dniDueño, "Error", JOptionPane.ERROR_MESSAGE);
            return;
        }

        Vehiculo actualizado = new Vehiculo(placa, anio, color, kilometraje, propietario);
        boolean ok = gestorVehiculos.actualizar(actualizado);

        if (ok) {
            JOptionPane.showMessageDialog(this, "Vehículo actualizado exitosamente.", "Éxito", JOptionPane.INFORMATION_MESSAGE);
            cargarTabla();
            limpiarFormulario();

        }
    }

    private void darDeBajaVehiculo() {
        String placa = txtPlaca.getText().trim();
        if (placa.isEmpty()) {
            JOptionPane.showMessageDialog(this, "Seleccione o ingrese la placa a dar de baja.", "Validación", JOptionPane.WARNING_MESSAGE);
            return;
        }

        Vehiculo v = gestorVehiculos.leer(placa);
        if (v == null) {
            JOptionPane.showMessageDialog(this, "No se encontró un vehículo activo con la placa indicada.", "Error", JOptionPane.ERROR_MESSAGE);
            return;
        }

        // El sistema verifica que no tenga órdenes activas
        if (gestorOrdenes != null && gestorOrdenes.tieneOrdenActiva(v)) {
            JOptionPane.showMessageDialog(this,
                    "No se puede dar de baja al vehículo con placa " + placa + 
                    " porque tiene una orden de trabajo activa en el taller.",
                    "Baja Rechazada",
                    JOptionPane.WARNING_MESSAGE);
            return;
        }

        int confirmacion = JOptionPane.showConfirmDialog(this,
                "¿Está seguro de dar de baja al vehículo con placa " + placa + "?",
                "Confirmar Baja Lógica",
                JOptionPane.YES_NO_OPTION);

        if (confirmacion == JOptionPane.YES_OPTION) {
            boolean dadoDeBaja = gestorVehiculos.darDeBaja(placa);
            if (dadoDeBaja) {
                JOptionPane.showMessageDialog(this, "Vehículo dado de baja correctamente.", "Éxito", JOptionPane.INFORMATION_MESSAGE);
                cargarTabla();
                limpiarFormulario();
            } else {
                JOptionPane.showMessageDialog(this, "No se pudo procesar la baja del vehículo.", "Error", JOptionPane.ERROR_MESSAGE);
            }
        }
    }

    private void buscarVehiculo() {
        String criterio = txtBuscarPlaca.getText().trim();
        if (criterio.isEmpty()) {
            cargarTabla();
            return;
        }

        // 1. Intentar buscar por placa exacta
        Vehiculo v = gestorVehiculos.leer(criterio);
        modeloTabla.setRowCount(0);
        if (v != null) {
            modeloTabla.addRow(new Object[]{
                    v.getPlaca(),
                    v.getAnio(),
                    v.getColor(),
                    v.getKilometraje(),
                    v.getPropietario() != null ? (v.getPropietario().getDni() + " - " + v.getPropietario().getNombres()) : "Sin dueño",
                    v.isActivo() ? "Activo" : "Inactivo"
            });
            return;
        }

        // 2. Intentar buscar si el criterio es el DNI de un cliente (RF-06)
        Cliente c = gestorClientes.leer(criterio);
        if (c != null) {
            Vehiculo[] listaCliente = gestorVehiculos.listarPorCliente(c);
            if (listaCliente.length > 0) {
                for (Vehiculo veh : listaCliente) {
                    modeloTabla.addRow(new Object[]{
                            veh.getPlaca(),
                            veh.getAnio(),
                            veh.getColor(),
                            veh.getKilometraje(),
                            veh.getPropietario() != null ? (veh.getPropietario().getDni() + " - " + veh.getPropietario().getNombres()) : "Sin dueño",
                            veh.isActivo() ? "Activo" : "Inactivo"
                    });
                }
                return;
            }
        }

        JOptionPane.showMessageDialog(this, "No se encontró ningún vehículo activo por placa o DNI de cliente: " + criterio, "Búsqueda", JOptionPane.INFORMATION_MESSAGE);
    }

    public void cargarTabla() {
        modeloTabla.setRowCount(0);
        Vehiculo[] lista = gestorVehiculos.listar();
        for (Vehiculo v : lista) {
            modeloTabla.addRow(new Object[]{
                    v.getPlaca(),
                    v.getAnio(),
                    v.getColor(),
                    v.getKilometraje(),
                    v.getPropietario() != null ? (v.getPropietario().getDni() + " - " + v.getPropietario().getNombres()) : "Sin dueño",
                    v.isActivo() ? "Activo" : "Inactivo"
            });
        }
    }

    private void cargarFormularioDesdeTabla() {
        int fila = tablaVehiculos.getSelectedRow();
        if (fila >= 0) {
            txtPlaca.setText(modeloTabla.getValueAt(fila, 0).toString());
            txtAnio.setText(modeloTabla.getValueAt(fila, 1).toString());
            txtColor.setText(modeloTabla.getValueAt(fila, 2).toString());
            txtKilometraje.setText(modeloTabla.getValueAt(fila, 3).toString());

            String propInfo = modeloTabla.getValueAt(fila, 4).toString();
            if (propInfo.contains(" - ")) {
                txtDniPropietario.setText(propInfo.split(" - ")[0]);
            }
        }
    }

    private void limpiarFormulario() {
        txtPlaca.setText("");
        txtAnio.setText("");
        txtColor.setText("");
        txtKilometraje.setText("");
        txtDniPropietario.setText("");
        tablaVehiculos.clearSelection();
    }

        itemSalir.addActionListener(e -> System.exit(0));

        menuArchivo.add(itemGuardar);
        menuArchivo.addSeparator();
        menuArchivo.add(itemSalir);
        menuBar.add(menuArchivo);
        setJMenuBar(menuBar);

        // 4. Configurar dimensiones y visualización de la ventana
        setTitle("Sistema de Gestión de Taller Mecánico - Algorítmica II");
        setSize(1040, 720);
        setMinimumSize(new Dimension(900, 600));

        getContentPane().removeAll();
        getContentPane().setLayout(new BorderLayout());
        getContentPane().add(pestañas, BorderLayout.CENTER);
        revalidate();
        repaint();
    }

    private void guardarDatosEnDisco() {
        try {
            GestorArchivos.guardarClientes(gestorClientes);
            GestorArchivos.guardarCatalogo(catalogo);
            GestorArchivos.guardarVehiculos(gestorVehiculos);
            GestorArchivos.guardarOrdenes(gestorOrdenes);
            GestorArchivos.guardarFacturas(historialFacturas);

            JOptionPane.showMessageDialog(this,
                    "¡Todos los datos han sido guardados exitosamente en la carpeta 'datos/'!",
                    "Guardado Exitoso",
                    JOptionPane.INFORMATION_MESSAGE);
        } catch (Exception ex) {
            JOptionPane.showMessageDialog(this,
                    "Ocurrió un error al guardar los archivos: " + ex.getMessage(),
                    "Error al Guardar",
                    JOptionPane.ERROR_MESSAGE);
        }
    }
